cmake_minimum_required(VERSION 3.26)
project(F4FftKernels LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CheckCSourceCompiles)
check_c_source_compiles("
#if !defined(_M_X64) && !defined(__x86_64__)
#error AVX2 archive requires x86-64
#endif
int main(void) { return 0; }" F4_TARGET_X64)

add_library(ngene_f4_fft_kernels STATIC ci8_fft_scalar.c)
target_include_directories(ngene_f4_fft_kernels PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(F4_TARGET_X64)
    add_library(ngene_f4_fft_avx2 OBJECT ci8_fft_avx2.c ci8_fft_compact_avx2.c)
    if(MSVC)
        target_compile_options(ngene_f4_fft_avx2 PRIVATE /arch:AVX2)
    else()
        target_compile_options(ngene_f4_fft_avx2 PRIVATE -mavx2)
    endif()
    target_sources(ngene_f4_fft_kernels PRIVATE $<TARGET_OBJECTS:ngene_f4_fft_avx2>)
endif()
set_property(TARGET ngene_f4_fft_kernels PROPERTY F4_HAS_AVX2 "${F4_TARGET_X64}")

add_executable(ngene_f4_fft_lab ci8_fft_lab.c)
target_link_libraries(ngene_f4_fft_lab PRIVATE ngene_f4_fft_kernels)
target_compile_definitions(ngene_f4_fft_lab PRIVATE CI8_LAB_LINK_KERNELS)
if(F4_TARGET_X64)
    target_include_directories(ngene_f4_fft_lab PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../src")
    target_compile_definitions(ngene_f4_fft_lab PRIVATE CI8_LAB_AVX2 CI8_LAB_COMPACT_AVX2 CI8_LAB_BUFFERED)
endif()
if(MSVC)
    target_compile_definitions(ngene_f4_fft_lab PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(ngene_f4_fft_lab PRIVATE /W4 /WX)
    target_compile_options(ngene_f4_fft_kernels PRIVATE /W4 /WX)
    if(TARGET ngene_f4_fft_avx2)
        target_compile_options(ngene_f4_fft_avx2 PRIVATE /W4 /WX)
    endif()
elseif(NOT WIN32)
    target_link_libraries(ngene_f4_fft_lab PRIVATE m)
endif()
install(TARGETS ngene_f4_fft_kernels ARCHIVE DESTINATION lib)
install(FILES ci8_fft_kernels.h DESTINATION include)

cmake_minimum_required(VERSION 3.26)
include(cmake/Dependencies.cmake)
ngene_prepare_dependencies()
project(NGene VERSION 0.6.3 LANGUAGES C)
option(NGENE_REQUIRE_MSVC "Require Microsoft C compiler for native Windows presets" OFF)
if(NGENE_REQUIRE_MSVC AND (NOT WIN32 OR NOT CMAKE_C_COMPILER_ID STREQUAL "MSVC"))
    message(FATAL_ERROR "This preset requires native MSVC. Use a compatible installed Visual Studio generator; no CMakeLists.txt edits are needed.")
endif()
if(POLICY CMP0144)
    cmake_policy(SET CMP0144 NEW)
endif()
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Windows|Linux|Darwin)$")
    message(FATAL_ERROR "This increment supports Windows, Linux and macOS targets")
endif()
option(NGENE_WITH_HACKRF "Build native libhackrf adapter" OFF)
option(NGENE_WITH_WEB "Build local web GUI and controller" ON)
option(NGENE_BUILD_TESTS "Build C contract/integration tests" ON)
option(NGENE_SANITIZERS "Address and undefined-behaviour sanitizers on GCC/Clang" OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(ThirdParty)
include(TargetArchitecture)
configure_file(cmake/build.h.in generated/ngene/build.h @ONLY)
find_package(Threads REQUIRED)
if(APPLE)
    add_compile_definitions(_DARWIN_C_SOURCE)
endif()
add_library(ngene_core
    src/core/error.c src/core/receiver.c src/core/stream_info.c src/config/config.c
    src/platform/platform.c src/platform/files.c src/stream/ring.c
    src/frontend/frontend.c src/frontend/file.c
    src/conditioning/bypass.c src/conditioning/dc.c src/dsp/sample_ops.c src/recording/recording.c src/recording/metadata.c)
target_include_directories(ngene_core PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated" PRIVATE src)
target_link_libraries(ngene_core PUBLIC Threads::Threads)
if(WIN32)
    target_link_libraries(ngene_core PRIVATE bcrypt)
    target_compile_definitions(ngene_core PRIVATE WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_definitions(ngene_core PRIVATE _POSIX_C_SOURCE=200809L _XOPEN_SOURCE=700 _FILE_OFFSET_BITS=64)
    target_link_libraries(ngene_core PUBLIC m)
endif()
if(MSVC)
    target_compile_options(ngene_core PRIVATE "$<$<COMPILE_LANGUAGE:C>:/W4;/WX;/utf-8;/external:W0>")
else()
    target_compile_options(ngene_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    if(NGENE_SANITIZERS)
        target_compile_options(ngene_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(ngene_core PUBLIC -fsanitize=address,undefined)
    endif()
endif()
if(NGENE_WITH_HACKRF)
    include(HackRFProvider)
    ngene_resolve_hackrf()
    target_sources(ngene_core PRIVATE src/frontend/hackrf.c)
    target_compile_definitions(ngene_core PRIVATE NGENE_HAVE_HACKRF=1)
    target_link_libraries(ngene_core PRIVATE HackRF::HackRF)
endif()
# Compiler drivers with an argument prefix (e.g. cross wrappers) need it for ASM too.
if(CMAKE_C_COMPILER_ARG1 AND NOT CMAKE_ASM_COMPILER_ARG1)
    set(CMAKE_ASM_COMPILER_ARG1 "${CMAKE_C_COMPILER_ARG1}" CACHE STRING "Assembler driver arguments" FORCE)
endif()
if(CMAKE_C_COMPILER_TARGET AND NOT CMAKE_ASM_COMPILER_TARGET)
    set(CMAKE_ASM_COMPILER_TARGET "${CMAKE_C_COMPILER_TARGET}")
endif()
option(NGENE_SAMPLE_REFERENCE "Use scalar sample primitives for validation" OFF)
if(NGENE_SAMPLE_REFERENCE)
    target_compile_definitions(ngene_core PRIVATE NGENE_SAMPLE_REFERENCE=1)
endif()
option(NGENE_FFT_REFERENCE "Explicit scalar FFT validation backend" OFF)
target_sources(ngene_core PRIVATE src/dsp/fft/fft.c src/analysis/analysis.c)
if(NGENE_FFT_REFERENCE)
    target_compile_definitions(ngene_core PRIVATE NGENE_FFT_BACKEND="C-reference")
elseif(NGENE_ARCH STREQUAL "x86_64")
    if(MSVC AND NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
        enable_language(ASM_MASM)
        target_sources(ngene_core PRIVATE src/dsp/x86/fft_sse2.asm)
    else()
        enable_language(ASM)
        target_sources(ngene_core PRIVATE src/dsp/x86/fft_sse2.S)
    endif()
    target_compile_definitions(ngene_core PRIVATE NGENE_FFT_ASSEMBLY=1 NGENE_FFT_BACKEND="SSE2-assembly")
elseif(NGENE_ARCH STREQUAL "arm64")
    if(MSVC AND NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
        enable_language(ASM_MARMASM)
        target_sources(ngene_core PRIVATE src/dsp/arm/fft_neon.asm)
    else()
        enable_language(ASM)
        target_sources(ngene_core PRIVATE src/dsp/arm/fft_neon.S)
    endif()
    target_compile_definitions(ngene_core PRIVATE NGENE_FFT_ASSEMBLY=1 NGENE_FFT_BACKEND="NEON-assembly")
else()
    message(FATAL_ERROR "Assembly FFT currently targets x86-64 and AArch64; for 32-bit validation select NGENE_FFT_REFERENCE")
endif()
add_executable(ngene apps/main.c)
target_link_libraries(ngene PRIVATE ngene_core)
if(NGENE_WITH_WEB)
    add_library(ngene_control src/control/control.c)
    target_include_directories(ngene_control PUBLIC include PRIVATE src)
    target_link_libraries(ngene_control PUBLIC ngene_core)
    if(WIN32)
        target_link_libraries(ngene_control PRIVATE bcrypt)
        target_compile_definitions(ngene_control PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_definitions(ngene_control PRIVATE _XOPEN_SOURCE=700 _FILE_OFFSET_BITS=64)
    endif()
    add_library(ngene_http STATIC third_party/civetweb/src/civetweb.c third_party/cjson/cJSON.c)
    ngene_external_target(ngene_http)
    target_include_directories(ngene_http SYSTEM PUBLIC third_party/civetweb/include third_party/cjson)
    target_compile_definitions(ngene_http PRIVATE USE_WEBSOCKET NO_SSL NO_CGI NO_CACHING CJSON_NESTING_LIMIT=32 NGENE_WS_MAX_PAYLOAD=4096 BUILD_DATE="NGene-vendor-v1.16")
    target_link_libraries(ngene_http PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
    if(WIN32)
        target_link_libraries(ngene_http PRIVATE ws2_32)
    else()
        target_compile_definitions(ngene_http PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64)
    endif()
    include(EmbedWeb)
    ngene_embed_web("${CMAKE_CURRENT_BINARY_DIR}/generated/web_assets.h")
    target_sources(ngene PRIVATE src/web/server.c "${CMAKE_CURRENT_BINARY_DIR}/generated/web_assets.c")
    target_include_directories(ngene PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(ngene PRIVATE NGENE_HAVE_WEB=1)
    target_link_libraries(ngene PRIVATE ngene_control ngene_http)
endif()
if(NGENE_BUILD_TESTS)
    enable_testing()
    add_test(NAME build_dependency_contract
        COMMAND "${CMAKE_COMMAND}" "-Dsource=${CMAKE_CURRENT_SOURCE_DIR}" "-Dwork=${CMAKE_CURRENT_BINARY_DIR}/build-contract" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/build_contract.cmake")
    if(UNIX)
        add_test(NAME runtime_staging_mock_contract
            COMMAND "${CMAKE_COMMAND}" "-Dsource=${CMAKE_CURRENT_SOURCE_DIR}" "-Dwork=${CMAKE_CURRENT_BINARY_DIR}/build-contract" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime_staging.cmake")
    endif()
    find_program(NGENE_POWERSHELL NAMES pwsh powershell)
    if(NGENE_POWERSHELL)
        add_test(NAME windows_verifier_helper_contract
            COMMAND "${NGENE_POWERSHELL}" -NoLogo -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/windows_verifier.ps1"
                -Source "${CMAKE_CURRENT_SOURCE_DIR}" -Work "${CMAKE_CURRENT_BINARY_DIR}/build-contract")
        add_test(NAME windows_developer_entry_contract
            COMMAND "${NGENE_POWERSHELL}" -NoLogo -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/windows_build.ps1"
                -Source "${CMAKE_CURRENT_SOURCE_DIR}" -Work "${CMAKE_CURRENT_BINARY_DIR}/build-contract")
    endif()
    add_executable(ngene_replay_tests tests/test_replay.c)
    target_include_directories(ngene_replay_tests PRIVATE src)
    target_link_libraries(ngene_replay_tests PRIVATE ngene_core)
    add_test(NAME file_replay_contract COMMAND ngene_replay_tests)
    add_executable(ngene_sample_tests tests/test_samples.c tests/fixtures.c)
    target_include_directories(ngene_sample_tests PRIVATE src)
    target_link_libraries(ngene_sample_tests PRIVATE ngene_core)
    add_test(NAME sample_math_and_dc COMMAND ngene_sample_tests)
    add_executable(ngene_analysis_tests tests/test_analysis.c)
    target_include_directories(ngene_analysis_tests PRIVATE src)
    target_link_libraries(ngene_analysis_tests PRIVATE ngene_core)
    add_test(NAME analysis_worker_contract COMMAND ngene_analysis_tests)
    add_executable(ngene_fft_tests tests/test_fft.c)
    target_include_directories(ngene_fft_tests PRIVATE src)
    target_link_libraries(ngene_fft_tests PRIVATE ngene_core)
    add_test(NAME fft_integer_contract COMMAND ngene_fft_tests)
    add_executable(ngene_tests tests/test_core.c)
    target_include_directories(ngene_tests PRIVATE src)
    if(NOT WIN32)
        target_compile_definitions(ngene_tests PRIVATE _POSIX_C_SOURCE=200809L)
    endif()
    target_link_libraries(ngene_tests PRIVATE ngene_core)
    add_test(NAME core_contracts COMMAND ngene_tests)

    add_test(NAME cli_help COMMAND ngene --help)
    add_test(NAME cli_version COMMAND ngene --version)
    if(WIN32 AND NGENE_WITH_HACKRF)
        add_test(NAME cli_hackrf_devices COMMAND ngene --list-devices)
    endif()
    set_tests_properties(core_contracts PROPERTIES TIMEOUT 30)
    if(NGENE_WITH_WEB)
        add_executable(ngene_control_tests tests/test_control.c tests/fixtures.c)
        target_include_directories(ngene_control_tests PRIVATE src)
        target_link_libraries(ngene_control_tests PRIVATE ngene_control)
        add_test(NAME controller_lifecycle COMMAND ngene_control_tests)
        set_tests_properties(controller_lifecycle PROPERTIES TIMEOUT 30)
    endif()
    if(NGENE_WITH_HACKRF AND UNIX)
        add_executable(ngene_hackrf_tests tests/test_hackrf.c tests/fixtures.c src/frontend/hackrf.c)
        target_include_directories(ngene_hackrf_tests PRIVATE src)
        target_include_directories(ngene_hackrf_tests SYSTEM PRIVATE "${HACKRF_INCLUDE_DIR}")
        target_compile_definitions(ngene_hackrf_tests PRIVATE _POSIX_C_SOURCE=200809L)
        target_link_libraries(ngene_hackrf_tests PRIVATE ngene_core)
        add_test(NAME native_hackrf_contract COMMAND ngene_hackrf_tests)
        set_tests_properties(native_hackrf_contract PROPERTIES TIMEOUT 20)
    endif()
endif()
# Keep upstream dependencies outside first-party warning policy.
foreach(target ngene ngene_hackrf_tests ngene_replay_tests ngene_control ngene_tests ngene_sample_tests ngene_analysis_tests ngene_fft_tests ngene_control_tests)
    if(TARGET ${target})
        if(MSVC)
            target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
            target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:/W4;/WX;/utf-8;/external:W0>")
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
        endif()
    endif()
endforeach()
# Discover both configurations' CRT files, but install only the DLLs staged with the
# selected binary. Debug runtime files are for local development, not redistribution.
if(MSVC)
    set(CMAKE_INSTALL_DEBUG_LIBRARIES TRUE)
    set(CMAKE_INSTALL_UCRT_LIBRARIES TRUE)
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS_SKIP TRUE)
endif()
include(InstallRequiredSystemLibraries)
include(RuntimeDependencies)
foreach(runtime ngene ngene_replay_tests ngene_tests ngene_sample_tests ngene_analysis_tests ngene_fft_tests ngene_control_tests)
    if(TARGET ${runtime})
        ngene_stage_runtime(${runtime})
    endif()
endforeach()
if(TARGET ngene_hackrf_native)
    if(APPLE)
        set_target_properties(ngene PROPERTIES INSTALL_RPATH "@loader_path/../lib")
        set_target_properties(ngene_hackrf_native PROPERTIES INSTALL_NAME_DIR "@rpath")
    elseif(UNIX)
        set_target_properties(ngene PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
    endif()
endif()
install(TARGETS ngene RUNTIME DESTINATION bin)
install(FILES third_party/civetweb/LICENSE.md DESTINATION share/ngene/licenses RENAME CivetWeb-LICENSE.md)
install(FILES third_party/cjson/LICENSE DESTINATION share/ngene/licenses RENAME cJSON-LICENSE)
install(FILES README.md docs/WINDOWS.md docs/PROVENANCE.md DESTINATION share/ngene/docs)
if(WIN32)
    install(DIRECTORY "$<TARGET_FILE_DIR:ngene>/" DESTINATION bin FILES_MATCHING PATTERN "*.dll")
endif()
set(CPACK_PACKAGE_NAME "NGene")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_GENERATOR "ZIP")
include(CPack)
install(DIRECTORY config/ DESTINATION share/ngene/config)
message(STATUS "NGene target ${CMAKE_SYSTEM_NAME}/${NGENE_ARCH}; native HackRF=${NGENE_WITH_HACKRF}")

option(NGENE_WITH_DOCUMENTATION "Build Doxygen reference (optional development tool)" OFF)
if(NGENE_WITH_DOCUMENTATION)
    find_package(Doxygen REQUIRED)
    set(DOXYGEN_EXTRACT_STATIC YES)
    set(DOXYGEN_WARN_IF_UNDOCUMENTED YES)
    set(DOXYGEN_WARN_IF_DOC_ERROR YES)
    set(DOXYGEN_WARN_AS_ERROR YES)
    set(DOXYGEN_WARN_NO_PARAMDOC YES)
    set(DOXYGEN_GENERATE_LATEX NO)
    doxygen_add_docs(ngene_docs include src apps COMMENT "Generate NGene API and implementation reference")
endif()

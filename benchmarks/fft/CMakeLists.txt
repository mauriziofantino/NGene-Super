cmake_minimum_required(VERSION 3.26)
project(NGeneFftBenchmark LANGUAGES C)

# This development-only project uses the production FFT sources, without SDR/GUI dependencies.
get_filename_component(NGENE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(NGENE_FFT_SOURCE_ROOT "${NGENE_ROOT}" CACHE PATH "Production source root; optionally an unmodified older checkout for before/after measurements")
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(NGENE_LEGACY_ROOT "" CACHE PATH "Optional untouched original NGene source root (contains asmFolder/fft.c)")
add_executable(ngene_fft_benchmark benchmark.c
    "${NGENE_FFT_SOURCE_ROOT}/src/dsp/fft/fft.c" "${NGENE_ROOT}/src/core/error.c")
target_include_directories(ngene_fft_benchmark PRIVATE "${NGENE_ROOT}/include" "${NGENE_ROOT}/src")
include("${NGENE_ROOT}/cmake/TargetArchitecture.cmake")
if(NGENE_ARCH STREQUAL "x86_64")
    if(MSVC)
        enable_language(ASM_MASM)
        target_sources(ngene_fft_benchmark PRIVATE "${NGENE_FFT_SOURCE_ROOT}/src/dsp/x86/fft_sse2.asm")
    else()
        enable_language(ASM)
        target_sources(ngene_fft_benchmark PRIVATE "${NGENE_FFT_SOURCE_ROOT}/src/dsp/x86/fft_sse2.S")
    endif()
    target_compile_definitions(ngene_fft_benchmark PRIVATE NGENE_FFT_ASSEMBLY=1 NGENE_FFT_BACKEND="SSE2-assembly")
elseif(NGENE_ARCH STREQUAL "arm64")
    if(MSVC)
        enable_language(ASM_MARMASM)
        target_sources(ngene_fft_benchmark PRIVATE "${NGENE_FFT_SOURCE_ROOT}/src/dsp/arm/fft_neon.asm")
    else()
        enable_language(ASM)
        target_sources(ngene_fft_benchmark PRIVATE "${NGENE_FFT_SOURCE_ROOT}/src/dsp/arm/fft_neon.S")
    endif()
    target_compile_definitions(ngene_fft_benchmark PRIVATE NGENE_FFT_ASSEMBLY=1 NGENE_FFT_BACKEND="NEON-assembly")
else()
    message(FATAL_ERROR "Benchmark supports x86-64 and AArch64 production assembly")
endif()
if(NOT WIN32)
    target_compile_definitions(ngene_fft_benchmark PRIVATE _POSIX_C_SOURCE=200809L)
    target_link_libraries(ngene_fft_benchmark PRIVATE m)
endif()
if(MSVC)
    target_compile_definitions(ngene_fft_benchmark PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(ngene_fft_benchmark PRIVATE "$<$<COMPILE_LANGUAGE:C>:/W4;/WX>")
    target_compile_options(ngene_fft_benchmark PRIVATE "$<$<AND:$<COMPILE_LANGUAGE:C>,$<CONFIG:Release>>:/O2;/Ob2>")
    target_link_options(ngene_fft_benchmark PRIVATE /INCREMENTAL:NO)
else()
    target_compile_options(ngene_fft_benchmark PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Wshadow;-Werror>")
    target_compile_options(ngene_fft_benchmark PRIVATE "$<$<AND:$<COMPILE_LANGUAGE:C>,$<CONFIG:Release>>:-O3>")
endif()
target_compile_definitions(ngene_fft_benchmark PRIVATE "$<$<CONFIG:Release>:NDEBUG>")
if(WIN32)
    target_compile_definitions(ngene_fft_benchmark PRIVATE WIN32 _WINDOWS)
endif()
target_compile_definitions(ngene_fft_benchmark PRIVATE NGENE_BENCH_COMPILER="${CMAKE_C_COMPILER_ID}-${CMAKE_C_COMPILER_VERSION}")

if(NGENE_LEGACY_ROOT)
    if(NOT NGENE_ARCH STREQUAL "x86_64")
        message(FATAL_ERROR "The original NGene FFT uses x86 SSE2; run the comparison on x86-64")
    endif()
    if(NOT CMAKE_C_COMPILER_ID MATCHES "^(GNU|Clang)$")
        message(FATAL_ERROR "The original GNU inline assembly requires GCC or Clang (MSVC cannot compile it)")
    endif()
    foreach(name fft.c fft.h simd.h)
        if(NOT EXISTS "${NGENE_LEGACY_ROOT}/asmFolder/${name}")
            message(FATAL_ERROR "Missing original asmFolder/${name} in NGENE_LEGACY_ROOT")
        endif()
        file(SHA256 "${NGENE_LEGACY_ROOT}/asmFolder/${name}" hash)
        message(STATUS "Original asmFolder/${name} SHA256=${hash}")
    endforeach()
    file(TO_CMAKE_PATH "${NGENE_LEGACY_ROOT}/asmFolder/fft.c" legacy_source)
    add_library(ngene_fft_legacy OBJECT legacy.c)
    if(MSVC)
        target_compile_options(ngene_fft_legacy PRIVATE "$<$<CONFIG:Release>:/O2;/Ob2>")
    else()
        target_compile_options(ngene_fft_legacy PRIVATE "$<$<CONFIG:Release>:-O3>")
    endif()
    target_compile_definitions(ngene_fft_legacy PRIVATE "$<$<CONFIG:Release>:NDEBUG>")
    if(WIN32)
        target_compile_definitions(ngene_fft_legacy PRIVATE WIN32 _WINDOWS)
    endif()
    target_include_directories(ngene_fft_legacy PRIVATE "${NGENE_LEGACY_ROOT}/asmFolder" "${NGENE_LEGACY_ROOT}/receiver" "${NGENE_LEGACY_ROOT}/utility")
    target_compile_definitions(ngene_fft_legacy PRIVATE NGENE_LEGACY_FFT_SOURCE="${legacy_source}" _USE_MATH_DEFINES)
    # Legacy macros omit XMM clobbers. Reserve their registers in GCC; disable auto-vectorization
    # with Clang and validate the compiled baseline before timing. No fast-math or native ISA.
    if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
        target_compile_options(ngene_fft_legacy PRIVATE -fno-tree-vectorize -ffixed-xmm0 -ffixed-xmm1 -ffixed-xmm2 -ffixed-xmm3)
    elseif(MSVC)
        target_compile_options(ngene_fft_legacy PRIVATE /clang:-fno-vectorize /clang:-fno-slp-vectorize /clang:-Wno-microsoft-include)
    else()
        target_compile_options(ngene_fft_legacy PRIVATE -fno-vectorize -fno-slp-vectorize)
    endif()
    if(NOT WIN32)
        target_compile_definitions(ngene_fft_legacy PRIVATE _POSIX_C_SOURCE=200809L _DEFAULT_SOURCE)
    endif()
    target_sources(ngene_fft_benchmark PRIVATE $<TARGET_OBJECTS:ngene_fft_legacy>)
    target_compile_definitions(ngene_fft_benchmark PRIVATE NGENE_BENCH_LEGACY=1)
endif()

cmake_minimum_required(VERSION 3.26)
project(NGeneFftBenchmark LANGUAGES C)

# This development-only project uses the production FFT sources, without SDR/GUI dependencies.
get_filename_component(NGENE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(NGENE_FFT_SOURCE_ROOT "${NGENE_ROOT}" CACHE PATH "Production source root; optionally an unmodified older checkout for before/after measurements")
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(NGENE_LEGACY_ROOT "" CACHE PATH "Optional untouched original NGene source root (contains asmFolder/fft.c)")
set(NGENE_FFT_BASELINE_ROOT "" CACHE PATH "Optional frozen EVO source root for an alternating same-process before/after comparison")
add_executable(ngene_fft_benchmark benchmark.c
    "${NGENE_FFT_SOURCE_ROOT}/src/dsp/fft/fft.c" "${NGENE_ROOT}/src/core/error.c")
target_include_directories(ngene_fft_benchmark PRIVATE "${NGENE_ROOT}/include" "${NGENE_ROOT}/src")
include("${NGENE_ROOT}/cmake/TargetArchitecture.cmake")
include("${NGENE_ROOT}/cmake/Q14Avx2.cmake")
ngene_add_q14_avx2(ngene_fft_benchmark "${NGENE_FFT_SOURCE_ROOT}")
include("${NGENE_ROOT}/cmake/FftBackend.cmake")
ngene_select_fft_backend("${NGENE_FFT_SOURCE_ROOT}" fft_kernel)
if(fft_kernel_assembler)
    enable_language(${fft_kernel_assembler})
endif()
target_compile_definitions(ngene_fft_benchmark PRIVATE ${fft_kernel_definitions}
    NGENE_BENCH_ARCH="${NGENE_ARCH}")
if(NGENE_ARCH STREQUAL "x86")
    set(fft_scalar_isa_option ${fft_kernel_scalar_isa_options})
    set_property(SOURCE "${NGENE_FFT_SOURCE_ROOT}/src/dsp/fft/fft.c" APPEND PROPERTY
        COMPILE_OPTIONS ${fft_scalar_isa_option})
endif()
if(fft_kernel_relative)
    set(fft_kernel "${NGENE_FFT_SOURCE_ROOT}/${fft_kernel_relative}")
    target_sources(ngene_fft_benchmark PRIVATE "${fft_kernel}")
    set_property(SOURCE "${fft_kernel}" APPEND PROPERTY COMPILE_OPTIONS ${fft_kernel_isa_options})
    file(SHA256 "${fft_kernel}" fft_kernel_hash)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${fft_kernel}")
else()
    set(fft_kernel_hash "none-scalar")
endif()
file(SHA256 "${NGENE_FFT_SOURCE_ROOT}/src/dsp/fft/fft.c" fft_source_hash)
message(STATUS "Current src/dsp/fft/fft.c SHA256=${fft_source_hash}")
message(STATUS "Current ${fft_kernel_relative} SHA256=${fft_kernel_hash}")
target_compile_definitions(ngene_fft_benchmark PRIVATE
    NGENE_BENCH_SOURCE_C_SHA256="${fft_source_hash}"
    NGENE_BENCH_SOURCE_KERNEL_SHA256="${fft_kernel_hash}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${NGENE_FFT_SOURCE_ROOT}/src/dsp/fft/fft.c")
if(NOT WIN32)
    target_compile_definitions(ngene_fft_benchmark PRIVATE _POSIX_C_SOURCE=200809L)
    target_link_libraries(ngene_fft_benchmark PRIVATE m)
endif()
if(MSVC)
    target_compile_definitions(ngene_fft_benchmark PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(ngene_fft_benchmark PRIVATE "$<$<COMPILE_LANGUAGE:C>:/W4;/WX>")
    target_compile_options(ngene_fft_benchmark PRIVATE "$<$<AND:$<COMPILE_LANGUAGE:C>,$<CONFIG:Release>>:/O2;/Ob2>")
    target_link_options(ngene_fft_benchmark PRIVATE /INCREMENTAL:NO)
else()
    target_compile_options(ngene_fft_benchmark PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Wshadow;-Werror>")
    target_compile_options(ngene_fft_benchmark PRIVATE "$<$<AND:$<COMPILE_LANGUAGE:C>,$<CONFIG:Release>>:-O3>")
endif()
target_compile_definitions(ngene_fft_benchmark PRIVATE "$<$<CONFIG:Release>:NDEBUG>")
if(WIN32)
    target_compile_definitions(ngene_fft_benchmark PRIVATE WIN32 _WINDOWS)
endif()
target_compile_definitions(ngene_fft_benchmark PRIVATE NGENE_BENCH_COMPILER="${CMAKE_C_COMPILER_ID}-${CMAKE_C_COMPILER_VERSION}")

if(NGENE_FFT_BASELINE_ROOT)
    set(baseline_c "${NGENE_FFT_BASELINE_ROOT}/src/dsp/fft/fft.c")
    ngene_select_fft_backend("${NGENE_FFT_BASELINE_ROOT}" baseline_kernel)
    if(baseline_kernel_assembler)
        enable_language(${baseline_kernel_assembler})
    endif()
    set(baseline_sources "${baseline_c}")
    foreach(source "${baseline_c}")
        if(NOT EXISTS "${source}")
            message(FATAL_ERROR "Missing frozen baseline source: ${source}")
        endif()
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
    endforeach()
    file(SHA256 "${baseline_c}" baseline_c_hash)
    if(baseline_kernel_relative)
        set(baseline_kernel "${NGENE_FFT_BASELINE_ROOT}/${baseline_kernel_relative}")
        if(NOT EXISTS "${baseline_kernel}")
            message(FATAL_ERROR "Missing frozen baseline source: ${baseline_kernel}")
        endif()
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${baseline_kernel}")
        file(SHA256 "${baseline_kernel}" baseline_kernel_hash)
        get_filename_component(kernel_extension "${baseline_kernel}" EXT)
        if(kernel_extension STREQUAL ".c")
            # The C preprocessor below renames every public and internal kernel symbol.
            list(APPEND baseline_sources "${baseline_kernel}")
            set_property(SOURCE "${baseline_kernel}" APPEND PROPERTY COMPILE_OPTIONS
                ${baseline_kernel_isa_options})
        else()
            # Keep frozen assembly arithmetic untouched; only exported symbol names change.
            file(READ "${baseline_kernel}" baseline_assembly)
            foreach(symbol Butterflies FirstStages InputInRange SimdAvailable)
                string(REPLACE "NgFft${symbol}" "NgBenchBaseline${symbol}"
                    baseline_assembly "${baseline_assembly}")
            endforeach()
            set(renamed_baseline_kernel "${CMAKE_CURRENT_BINARY_DIR}/fft_baseline_kernel${kernel_extension}")
            file(WRITE "${renamed_baseline_kernel}" "${baseline_assembly}")
            list(APPEND baseline_sources "${renamed_baseline_kernel}")
        endif()
    else()
        set(baseline_kernel_hash "none-scalar")
    endif()
    message(STATUS "Before src/dsp/fft/fft.c SHA256=${baseline_c_hash}")
    message(STATUS "Before ${baseline_kernel_relative} SHA256=${baseline_kernel_hash}")
    add_library(ngene_fft_baseline OBJECT ${baseline_sources})
    if(NGENE_ARCH STREQUAL "x86")
        set_property(SOURCE "${baseline_c}" APPEND PROPERTY COMPILE_OPTIONS ${fft_scalar_isa_option})
    endif()
    # Compile before and after with the same compiler, flags, ISA and diagnostics.
    target_include_directories(ngene_fft_baseline PRIVATE
        "$<TARGET_PROPERTY:ngene_fft_benchmark,INCLUDE_DIRECTORIES>")
    target_compile_options(ngene_fft_baseline PRIVATE
        "$<TARGET_PROPERTY:ngene_fft_benchmark,COMPILE_OPTIONS>")
    target_compile_definitions(ngene_fft_baseline PRIVATE ${baseline_kernel_definitions}
        "$<$<CONFIG:Release>:NDEBUG>")
    if(WIN32)
        target_compile_definitions(ngene_fft_baseline PRIVATE WIN32 _WINDOWS _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_definitions(ngene_fft_baseline PRIVATE _POSIX_C_SOURCE=200809L)
    endif()
    foreach(symbol Create Destroy Execute Reference Backend Butterflies FirstStages InputInRange SimdAvailable)
        target_compile_definitions(ngene_fft_baseline PRIVATE
            "$<$<COMPILE_LANGUAGE:C>:NgFft${symbol}=NgBenchBaseline${symbol}>")
    endforeach()
    target_sources(ngene_fft_benchmark PRIVATE $<TARGET_OBJECTS:ngene_fft_baseline>)
    target_compile_definitions(ngene_fft_benchmark PRIVATE NGENE_BENCH_BASELINE=1
        NGENE_BENCH_BASELINE_C_SHA256="${baseline_c_hash}"
        NGENE_BENCH_BASELINE_KERNEL_SHA256="${baseline_kernel_hash}")
endif()

if(NGENE_LEGACY_ROOT)
    if(NOT NGENE_ARCH MATCHES "^(x86|x86_64)$")
        message(FATAL_ERROR "The original NGene FFT uses x86 SSE2; run the comparison on x86 or x86-64")
    endif()
    if(NOT CMAKE_C_COMPILER_ID MATCHES "^(GNU|Clang)$")
        message(FATAL_ERROR "The original GNU inline assembly requires GCC or Clang (MSVC cannot compile it)")
    endif()
    foreach(name fft.c fft.h simd.h)
        if(NOT EXISTS "${NGENE_LEGACY_ROOT}/asmFolder/${name}")
            message(FATAL_ERROR "Missing original asmFolder/${name} in NGENE_LEGACY_ROOT")
        endif()
        file(SHA256 "${NGENE_LEGACY_ROOT}/asmFolder/${name}" hash)
        message(STATUS "Original asmFolder/${name} SHA256=${hash}")
        if(name STREQUAL "fft.c")
            set(legacy_fft_c_hash "${hash}")
        elseif(name STREQUAL "fft.h")
            set(legacy_fft_h_hash "${hash}")
        else()
            set(legacy_simd_h_hash "${hash}")
        endif()
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
            "${NGENE_LEGACY_ROOT}/asmFolder/${name}")
    endforeach()
    file(TO_CMAKE_PATH "${NGENE_LEGACY_ROOT}/asmFolder/fft.c" legacy_source)
    add_library(ngene_fft_legacy OBJECT legacy.c)
    if(MSVC)
        target_compile_options(ngene_fft_legacy PRIVATE "$<$<CONFIG:Release>:/O2;/Ob2>")
    else()
        target_compile_options(ngene_fft_legacy PRIVATE "$<$<CONFIG:Release>:-O3>")
    endif()
    target_compile_definitions(ngene_fft_legacy PRIVATE "$<$<CONFIG:Release>:NDEBUG>")
    if(WIN32)
        target_compile_definitions(ngene_fft_legacy PRIVATE WIN32 _WINDOWS)
    endif()
    target_include_directories(ngene_fft_legacy PRIVATE "${NGENE_LEGACY_ROOT}/asmFolder" "${NGENE_LEGACY_ROOT}/receiver" "${NGENE_LEGACY_ROOT}/utility")
    target_compile_definitions(ngene_fft_legacy PRIVATE NGENE_LEGACY_FFT_SOURCE="${legacy_source}" _USE_MATH_DEFINES)
    if(NGENE_ARCH STREQUAL "x86")
        if(MSVC)
            target_compile_options(ngene_fft_legacy PRIVATE /arch:SSE2)
        else()
            target_compile_options(ngene_fft_legacy PRIVATE -msse2)
        endif()
    endif()
    # Legacy macros omit XMM clobbers. Reserve their registers in GCC; disable auto-vectorization
    # with Clang and validate the compiled baseline before timing. No fast-math or native ISA.
    if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
        target_compile_options(ngene_fft_legacy PRIVATE -fno-tree-vectorize -ffixed-xmm0 -ffixed-xmm1 -ffixed-xmm2 -ffixed-xmm3)
    elseif(MSVC)
        target_compile_options(ngene_fft_legacy PRIVATE /clang:-fno-vectorize /clang:-fno-slp-vectorize /clang:-Wno-microsoft-include)
    else()
        target_compile_options(ngene_fft_legacy PRIVATE -fno-vectorize -fno-slp-vectorize)
    endif()
    if(NOT WIN32)
        target_compile_definitions(ngene_fft_legacy PRIVATE _POSIX_C_SOURCE=200809L _DEFAULT_SOURCE)
    endif()
    target_sources(ngene_fft_benchmark PRIVATE $<TARGET_OBJECTS:ngene_fft_legacy>)
    target_compile_definitions(ngene_fft_benchmark PRIVATE NGENE_BENCH_LEGACY=1
        NGENE_BENCH_LEGACY_FFT_C_SHA256="${legacy_fft_c_hash}"
        NGENE_BENCH_LEGACY_FFT_H_SHA256="${legacy_fft_h_hash}"
        NGENE_BENCH_LEGACY_SIMD_H_SHA256="${legacy_simd_h_hash}")
endif()

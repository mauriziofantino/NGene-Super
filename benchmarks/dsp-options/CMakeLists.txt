cmake_minimum_required(VERSION 3.26)
project(NGeneDspOptions LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(Threads REQUIRED)
option(REFERENCE_ONLY "Disable explicit SIMD for the candidate kernels" OFF)
option(DSP_OPTIONS_AVX2 "Build the experimental x86 AVX2 dot products" OFF)
add_executable(dsp_options experiment.c "${ROOT}/src/dsp/gps_ca.c" "${ROOT}/src/core/error.c" "${ROOT}/src/platform/platform.c")
target_include_directories(dsp_options PRIVATE "${ROOT}/include" "${ROOT}/src")
target_link_libraries(dsp_options PRIVATE Threads::Threads)
if(DSP_OPTIONS_AVX2)
  if(REFERENCE_ONLY)
    message(FATAL_ERROR "DSP_OPTIONS_AVX2 requires REFERENCE_ONLY=OFF")
  endif()
  target_sources(dsp_options PRIVATE dot_avx2.c)
  target_compile_definitions(dsp_options PRIVATE DSP_OPTIONS_AVX2=1)
  if(MSVC)
    set_source_files_properties(dot_avx2.c PROPERTIES COMPILE_OPTIONS /arch:AVX2)
  else()
    set_source_files_properties(dot_avx2.c PROPERTIES COMPILE_OPTIONS -mavx2)
  endif()
endif()
if(REFERENCE_ONLY)
  target_compile_definitions(dsp_options PRIVATE REFERENCE_ONLY=1)
endif()
if(WIN32)
  target_compile_definitions(dsp_options PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(dsp_options PRIVATE bcrypt)
else()
  target_compile_definitions(dsp_options PRIVATE _POSIX_C_SOURCE=200809L)
endif()
if(MSVC)
  target_compile_options(dsp_options PRIVATE /O2 /W4 /WX /utf-8)
else()
  target_compile_options(dsp_options PRIVATE -O3 -Wall -Wextra -Werror)
  target_link_libraries(dsp_options PRIVATE m)
endif()
target_compile_definitions(dsp_options PRIVATE NGENE_EXPERIMENT_COMPILER="${CMAKE_C_COMPILER_ID}-${CMAKE_C_COMPILER_VERSION}")
